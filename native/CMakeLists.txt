cmake_minimum_required(VERSION 3.16)
project(QuartzNative VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_library(quartz_native STATIC src/tokenizer.c src/search_core.cpp)
target_include_directories(quartz_native PUBLIC include)
enable_testing()
add_executable(quartz_native_tests tests/core_tests.cpp)
target_link_libraries(quartz_native_tests PRIVATE quartz_native)
add_test(NAME quartz_native_tests COMMAND quartz_native_tests)
